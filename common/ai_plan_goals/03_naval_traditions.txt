# Makes naval powers value the heavy ship tradition paths and actually take them. Probably needs balancing.
# Every step on the octere/mega polyreme tradition paths carries blockade_efficiency = 0.05 as a helper to guide AI to take
# them, cuz modifiers are what An-bleed used to guide the  AI.
# Only countries with a level 3+ port value it, so landlocked and minor countries ignore those paths.
# Scale: 0.05 x 6000 = 300, about the same as +15% tax income for anyone else touching this.

naval_power_aimod = {
	trigger = {
		any_owned_province = {
			num_of_port_building >= 3
		}
	}

	blockade_efficiency = 6000
}
