seleukid_mission_02_syria = {#Finishing is 11 tasks (same as vanilla)
    header = "mission_image_seleukid"
    #icon = "seleukid_2"
    icon = "cde_greek_soldiers_marching"

    repeatable = no

    chance = {
        factor = 5
    }

    potential = {
        NOT = { has_variable = seleukid_mission_02_syria }
        has_dlc = "Heirs of Alexander"
        OR = {
            tag = SEL
            has_variable = sel_unique_mech
        }
        is_subject = no
        has_civil_war = no
        is_monarchy = yes
    }

    abort = {
        OR = {
            is_monarchy = no
            trigger_if = {
                limit = { is_ai = yes }
                NOT = {
                    any_owned_province = {
                        OR = {
                            is_in_region = syria_region
                            is_in_region = assyria_region
                            is_in_region = cappadocia_region
                            is_in_region = cilicia_region
                        }
                    }
                }
            }
            ai_mission_back_out_trigger = yes
        }
    }

    on_start = {
        hidden_effect = {
            start_mission_ai_effect = yes
            random_list = { # roll whether the AI may release subjects in this tree (25% yes) #this is further diluted by ai_will_do #this is further diluted by ai_will_do
                25 = { set_variable = { name = seleukid_syria_ai_release value = 1 } }
                75 = { set_variable = { name = seleukid_syria_ai_release value = 0 } }
            }
            seleukid_roll_mission_pace_effect = yes
            if = { # scale this tree's gate timers to the pace
                limit = { is_ai = yes }
                if = {
                    limit = { var:seleukid_mission_pace = 5 }
                    set_variable = { name = seleukid_syria_bypass_gate_1 days = 912 }
                    set_variable = { name = seleukid_syria_bypass_gate_2 days = 1825 }
                }
                else_if = {
                    limit = { var:seleukid_mission_pace = 10 }
                    set_variable = { name = seleukid_syria_bypass_gate_1 days = 1825 }
                    set_variable = { name = seleukid_syria_bypass_gate_2 days = 3650 }
                }
                else = {
                    set_variable = { name = seleukid_syria_bypass_gate_1 days = 2737 }
                    set_variable = { name = seleukid_syria_bypass_gate_2 days = 5475 }
                }
            }
            if = {
                limit = {
                    owns_or_subject_owns_area = syria_area
                }
                set_variable = seleukid_mission_02_owns_syria_start
            }
            if = {
                limit = {
                    owns_or_subject_owns_area = syria_cyrrestice_area
                }
                set_variable = seleukid_mission_02_owns_syria_cyrrestice_start
            }
            if = {
                limit = {
                    OR = {
                        NOT = { exists = c:OBE }
                        c:OBE = { has_land = no }
                        c:OBE = {
                            is_subject = yes
                            overlord = {
                                this = root
                            }
                        }
                    }
                }
                set_variable = skip_olbe
            }         
        }
    }

    on_abort = {
        seleukid_mission_02_variable_cleanup_effect = yes
        custom_tooltip = general_mission_cooldown_tt
        set_variable = {
            name = seleukid_mission_02_syria
            days = 7300
        }
    }

    on_completion = {
        complete_mission_effect = yes
