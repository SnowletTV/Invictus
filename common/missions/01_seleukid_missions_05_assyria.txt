seleukid_mission_05_assyria_armenia = {#Finishing is 9 tasks (same as vanilla)
    header = "mission_image_seleukid"
    icon = "seleukid_2"

    repeatable = no

    chance = {
        factor = 5
    }

    potential = {
        NOT = { has_variable = seleukid_mission_05_assyria_armenia }
        has_dlc = "Heirs of Alexander"
        OR = {
            tag = SEL
            has_variable = sel_unique_mech
        }
        is_subject = no
        has_civil_war = no
        is_monarchy = yes
        has_completed_mission = seleukid_mission_01_babylonia
        trigger_if = { #The AI owns enough land to start
            limit = { is_ai = yes }
            sel_assyria_northern_base_svalue >= 25
        }
    }

    abort = {
        OR = {
            is_monarchy = no
            ai_mission_back_out_trigger = yes
        }
    }

    on_start = {
        hidden_effect = {
            start_mission_ai_effect = yes
            random_list = { # roll whether the AI may release subjects in this tree (25% yes) #this is further diluted by ai_will_do #this is further diluted by ai_will_do
                25 = { set_variable = { name = seleukid_assyria_ai_release value = 1 } }
                75 = { set_variable = { name = seleukid_assyria_ai_release value = 0 } }
            }
            seleukid_roll_mission_pace_effect = yes
            if = { # scale this tree's gate timers to the pace
                limit = { is_ai = yes }
                if = {
                    limit = { var:seleukid_mission_pace = 5 }
                    set_variable = { name = seleukid_assyria_bypass_gate_1 days = 1824 }
                    set_variable = { name = seleukid_assyria_bypass_gate_2 days = 3650 }
                }
                else_if = {
                    limit = { var:seleukid_mission_pace = 10 }
                    set_variable = { name = seleukid_assyria_bypass_gate_1 days = 3650 }
                    set_variable = { name = seleukid_assyria_bypass_gate_2 days = 7300 }
                }
                else = {
                    set_variable = { name = seleukid_assyria_bypass_gate_1 days = 5474 }
                    set_variable = { name = seleukid_assyria_bypass_gate_2 days = 10950 }
                }
            }
        }
    }

    on_abort = {
        custom_tooltip = general_mission_cooldown_tt
        set_variable = {
            name = seleukid_mission_05_assyria_armenia
            days = 7300
        }
    }

    on_completion = {
        complete_mission_effect = yes
        seleukid_mission_02_variable_cleanup_effect = yes
        trigger_event = me_seleukid_05_assyria.500
    }
