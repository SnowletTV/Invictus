parisii_lutetia_mission = {
    header = "parisii_lutetia_mission_header"
    icon = "cde_gallic_chiefs_council"
    repeatable = no

    chance = {
        factor = 5000 # Viable AI prioritization factor
    }

    ai_chance = {
        factor = 5000
    }

    potential = {
        NOT = { has_variable = paris_mission_cooldown }
        OR = {
            tag = PRS
            AND = {
                primary_culture = sennonian
                owns = 2475 # Lutetia
            }

    }

        save_scope_as = paris_mission_scope
        start_mission_ai_effect = yes
        set_variable = { name = seine_forts_built value = 0 }

        ai_mission_back_out_trigger = yes # Required Invictus safety trigger
    }

    on_abort = {
        custom_tooltip = general_mission_cooldown_tt
        set_variable = {
            name = paris_mission_cooldown
            days = 7300
        }
    }

    on_completion = {
        complete_mission_effect = yes
    }
