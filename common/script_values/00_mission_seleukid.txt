# 5 - Assyria (tree 05)

# AI start gate: owned provinces across the northern regions the tree spans.
# Used in the series potential so the AI only picks the tree up with a real base to finish it.
sel_assyria_northern_base_svalue = {
	value = 0
	every_owned_province = {
		limit = {
			OR = {
				is_in_region = mesopotamia_region
				is_in_region = assyria_region
				is_in_region = armenia_region
			}
		}
		add = 1
	}
}

# Final gate: "most of Armenia" - provinces in armenia_region held by ROOT or a ROOT subject.
# Threshold set in the final task's allow (tunable).
sel_assyria_armenia_control_svalue = {
	value = 0
	every_province = {
		limit = {
			is_in_region = armenia_region
			OR = {
				owner = root
				owner = { is_subject_of = root }
			}
		}
		add = 1
	}
}
