# ==================================================================================================
# AI SHIP TYPE RATIO
# Counts are taken monthly by ai_update_ship_ratio and stored on the country
# ==================================================================================================

count_light_ships = {
    value = 0
    every_navy = {
        limit = {
            is_mercenary_unit = no
        }
        every_sub_unit = {
            limit = {
                OR = {
                    sub_unit_type = liburnian
                    sub_unit_type = trireme
                }
            }
            add = 1
        }
    }
}

count_medium_ships = {
    value = 0
    every_navy = {
        limit = {
            is_mercenary_unit = no
        }
        every_sub_unit = {
            limit = {
                OR = {
                    sub_unit_type = tetrere
                    sub_unit_type = hexere
                }
            }
            add = 1
        }
    }
}

count_heavy_ships = {
    value = 0
    every_navy = {
        limit = {
            is_mercenary_unit = no
        }
        every_sub_unit = {
            limit = {
                OR = {
                    sub_unit_type = octere
                    sub_unit_type = mega_galley
                }
            }
            add = 1
        }
    }
}

ship_total = {
    value = var:ship_light
    add = var:ship_medium
    add = var:ship_heavy
}

# Ratios: 3 mediums for every light ship, 5 mediums for every heavy ship
ships_mediums_per_light = 3
ships_medium_per_heavy = 5
# Below this many ships mediums are never blocked (lights have no minimum)
ship_ratio_min_fleet = 20

# >= 0 means there are already enough light ships for the current mediums (lights x 3 >= mediums)
ship_light_excess = {
    value = var:ship_light
    multiply = ships_mediums_per_light
    subtract = var:ship_medium
}

# >= 0 means there are already enough medium ships for the current heavies
ship_medium_excess = {
    value = var:ship_medium
    subtract = {
        value = var:ship_heavy
        multiply = ships_medium_per_heavy
    }
}

# ==================================================================================================
# AI WOOD IMPORT FOR NAVIES
# ==================================================================================================

# Regional powers (25+ territories) need at least this many own ships for the bigger cap
wood_import_min_ships = 15

# Smaller countries need at least this many pops in their capital territory
wood_import_min_capital_pops = 20

# Max wood routes created per country per year
wood_import_cap_big_navy = 2
wood_import_cap_other = 1

# State scope: highest port level in the state, capital state slightly preferred on ties
wood_import_state_priority = {
    value = 0
    if = {
        limit = { any_state_province = { num_of_port_building >= 5 } }
        value = 5
    }
    else_if = {
        limit = { any_state_province = { num_of_port_building >= 4 } }
        value = 4
    }
    else_if = {
        limit = { any_state_province = { num_of_port_building >= 3 } }
        value = 3
    }
    else_if = {
        limit = { any_state_province = { num_of_port_building >= 2 } }
        value = 2
    }
    else = {
        value = 1
    }
    if = {
        limit = { is_capital_state = yes }
        add = 0.5
    }
}
