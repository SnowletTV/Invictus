
galatian_horde_effect = {
	set_variable = {
		name = is_not_allowed_to_go_broke
		days = 9125
	}
	set_variable = {
		name = is_ignoring_ae
		days = 3650
	}
	create_character = {
		culture = current_ruler.culture
		religion = current_ruler.religion
		female = no
		age = adult_character_age_range_svalue
		save_scope_as = galatian_horde_commander
	}
	create_unit = {
		navy = no
		location = $LOC$
		commander = scope:galatian_horde_commander
		while = {
			count = $INF$
			add_subunit = light_infantry
		}
		while = {
			count = $CHA$
			add_subunit = chariots
		}
		add_food = $FOOD$
		add_morale = 2
	}
}


galatian_opportunist_wars_effect = {
	if = {
		limit = { scope:$DEFENDER$ = { war_with = c:MAC } }
		declare_war_with_wargoal = {
			war_goal = conquer_wargoal
			province = scope:$PROVINCE$
			target = c:MAC
		}
	}
	if = {
		limit = { scope:$DEFENDER$ = { war_with = c:TRE } }
		declare_war_with_wargoal = {
			war_goal = conquer_wargoal
			province = scope:$PROVINCE$
			target = c:TRE
		}
	}
	if = {
		limit = { scope:$DEFENDER$ = { war_with = c:SEL } }
		declare_war_with_wargoal = {
			war_goal = conquer_wargoal
			province = scope:$PROVINCE$
			target = c:SEL
		}
	}
	if = {
		limit = { scope:$DEFENDER$ = { war_with = c:EGY } }
		declare_war_with_wargoal = {
			war_goal = conquer_wargoal
			province = scope:$PROVINCE$
			target = c:EGY
		}
	}
}

galatian_peace_conquest_tail_effect = {
	set_capital = $CAPITAL$
	trigger_event = {
		id = galatian_invasion.44
		days = 1
	}
}

change_or_set_variable_effect = {
	if = {
		limit = { has_variable = $VAR$ }
		change_variable = {
			name = $VAR$
			add = 1
		}
	}
	else = {
		set_variable = {
			name = $VAR$
			value = 1
		}
	}
}

award_tournament_points_effect = {
	if = {
		limit = { has_variable = tournament_points }
		change_variable = {
			name = tournament_points
			add = $POINTS$
		}
	}
	else = {
		set_variable = {
			name = tournament_points
			value = $POINTS$
		}
	}
}
