# Rolls a fresh AI mission pace (5/10/15) each on_start; the tree reads it to set its bypass-gate timers.
seleukid_roll_mission_pace_effect = {
	if = {
		limit = {
			is_ai = yes
		}
		random_list = {
			50 = { set_variable = { name = seleukid_mission_pace value = 5 } }
			30 = { set_variable = { name = seleukid_mission_pace value = 10 } }
			20 = { set_variable = { name = seleukid_mission_pace value = 15 } }
		}
	}
}

# Province-scope helper. convert up to $PROVINCE_MAX$ leading non-Hellenic pops in the province, one in each neighbouring province we or our subjects hold, then schedule the polis-maturation event 5-25 years out.
seleukid_colony_seed_hellenization_effect = {
	ordered_pops_in_province = { 
		order_by = pop_hapiness
		limit = {
			OR = { pop_type = citizen  pop_type = nobles  pop_type = freemen }
			NOT = { pop_culture_group = hellenic }
			pop_religion = roman_pantheon
		}
		set_pop_culture = greco_persian
		max = $PROVINCE_MAX$
		check_range_bounds = no
	}
	ordered_pops_in_province = {
		order_by = pop_hapiness
		limit = {
			OR = { pop_type = citizen  pop_type = nobles  pop_type = freemen }
			NOT = { pop_culture_group = hellenic }
		}
		set_pop_culture = greco_persian
		max = 1
		check_range_bounds = no
	}
	every_neighbor_province = { # a touch in the surrounding country
		limit = {
			is_inhabitable = yes
			exists = owner
			OR = { owner = root  owner = { is_subject_of = root } }
		}
		ordered_pops_in_province = {
			order_by = pop_hapiness
			limit = {
				OR = { pop_type = citizen  pop_type = freemen }
				NOT = { pop_culture_group = hellenic }
			}
			set_pop_culture = greco_persian
			max = 1
			check_range_bounds = no
		}
	}
	trigger_event = { id = seleukid_colony.1  days = { 1825 9125 } }
}

# Province-scope: turn a province's pops toward Greco-Persian.
seleukid_greco_persian_convert_pops_effect = {
	every_pops_in_province = { # every culturally non-Hellenic pop of the Hellenic religion
		limit = {
			pop_religion = roman_pantheon
			NOT = { pop_culture_group = hellenic }
		}
		set_pop_culture = greco_persian
	}
	ordered_pops_in_province = { # plus a handful of the leading non-Hellenic pops, whatever their faith
		order_by = pop_hapiness
		limit = {
			OR = { pop_type = citizen  pop_type = nobles  pop_type = freemen }
			NOT = { pop_culture_group = hellenic }
		}
		set_pop_culture = greco_persian
		max = 5
		check_range_bounds = no
	}
}

# Province-scope: a heavy Hellenisation of one province. Every roman_pantheon non-Hellenic pop converts at once, plus a large number of other non-Hellenic pops of any faith, more in cities than in settlements.
seleukid_greco_persian_convert_province_heavy_effect = {
	every_pops_in_province = {
		limit = {
			pop_religion = roman_pantheon
			NOT = { pop_culture_group = hellenic }
		}
		set_pop_culture = greco_persian
	}
	if = {
		limit = { has_city_status = yes }
		ordered_pops_in_province = {
			order_by = pop_hapiness
			limit = { NOT = { pop_culture_group = hellenic } }
			set_pop_culture = greco_persian
			max = 6
			check_range_bounds = no
		}
	}
	else = {
		ordered_pops_in_province = {
			order_by = pop_hapiness
			limit = { NOT = { pop_culture_group = hellenic } }
			set_pop_culture = greco_persian
			max = 2
			check_range_bounds = no
		}
	}
}

# Country-scope (run ON the country being Hellenised, e.g. a Persis subject): it takes Greco-Persian as its primary culture, its ruling house turns Greco-Persian, and every one of its provinces is heavily converted (cities more than settlements).
seleukid_greco_persianize_country_effect = {
	set_primary_culture = greco_persian
	seleukid_greco_persian_characters_effect = yes
	every_owned_province = {
		seleukid_greco_persian_convert_province_heavy_effect = yes
	}
}

# Country-scope: integrate Greco-Persian for the AI, then convert Median pops
seleukid_hellenize_media_effect = {
	if = { # the AI takes up Greco-Persian as an integrated culture if it has not already
		limit = {
			is_ai = yes
			NOT = { any_integrated_culture = { is_culture = greco_persian } }
			any_country_culture = { THIS.culture = culture:greco_persian }
		}
		random_country_culture = {
			limit = { THIS.culture = culture:greco_persian }
			ROOT = { integrate_country_culture = PREV }
		}
	}
	every_owned_province = {
		limit = { is_in_region = media_region }
		seleukid_greco_persian_convert_pops_effect = yes
	}
	every_subject = {
		limit = { capital_scope = { is_in_region = media_region } }
		random_list = {
			60 = { set_primary_culture = greco_persian seleukid_greco_persian_ruler_characters_effect = yes }
			40 = { add_stability = 5 } #nothing
		}
		set_primary_culture = greco_persian
		seleukid_greco_persian_characters_effect = yes
		every_owned_province = {
			limit = { is_in_region = media_region }
			seleukid_greco_persian_convert_pops_effect = yes
		}
	}
}

# Some non-royal chartacters become Greco-Persian.
seleukid_greco_persian_characters_effect = {
	ordered_character = { # some of the children
		limit = {
			is_ruler = no
			is_successor = no
			is_adult = no
			is_primary_heir = no
		}
		max = 5
		check_range_bounds = no
		order_by = power_base
		set_culture = greco_persian
	}
	ordered_character = { # a few prominent adults
		limit = {
			is_ruler = no
			from_ruler_family = no
			is_adult = yes
		}
		max = 3
		order_by = prominence
		check_range_bounds = no
		set_culture = greco_persian
	}
}

# Ruling family becomes Greco-Persian.
seleukid_greco_persian_ruler_characters_effect = {
	every_character = {
		limit = {
			from_ruler_family = yes
		}
		set_culture = greco_persian
	}
	ordered_character = { # a few prominent adults
		limit = {
			is_ruler = no
			from_ruler_family = no
			is_adult = yes
		}
		max = 5
		order_by = prominence
		check_range_bounds = no
		set_culture = greco_persian
	}
}
