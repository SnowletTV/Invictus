country_decisions = {
	
	# Form the Vandal Kingdom
	form_vandalia_africa = {
		
		potential = {
			has_global_variable = ahistorical_formables_variable
			num_of_cities >= 1
			primary_culture = vandal
			is_tribal = yes
			NOT = {
				tag = VAN
			}
			NOR = {
				is_tier_2_formable_trigger = yes
				is_endgame_tag_trigger = yes
			}
			OR = {
				is_ai = no
				num_of_cities <= 12
			}
			capital_scope = {
				OR = {
					is_in_region = africa_region
					is_in_region = numidia_region
					is_in_region = mauretainia_region
					is_in_region = atlas_region
				}
			}
		}
		
		highlight = {
			scope:province = {
				OR = {
					is_in_region = africa_region
					is_in_region = numidia_region
				}
			}
		}
		
		allow = {
			custom_tooltip = {
				text = vandalia_africa_not_exists
				
			}
			can_form_nation_trigger = yes
			AND = {
				owns_or_subject_owns_region = africa_region
				owns_or_subject_owns_region = numidia_region
			}
			custom_tooltip = {
				text = formable_vandalia_africa_capital
				capital_scope = {
					OR = {
						is_in_region = africa_region
						is_in_region = numidia_region
						is_in_region = mauretainia_region
						is_in_region = atlas_region
					}
				}
			}
		}
		
		effect = {
			custom_tooltip = "tier_2_formable_tooltip"
			if = {
				limit = { is_tribal = yes }
				add_country_modifier = {
					name = civilized_tribe
					duration = 5475
				}
				change_government = tribal_federation
				switch_government_type_event_clearup_effect = yes
				change_country_name = VANDALIA_AFRICA_NAME
			}
			else_if = {
				limit = { is_republic = yes }
				change_country_name = VANDALIA_AFRICA_REPUBLIC_NAME
			}
			else = {
				change_country_name = VANDALIA_AFRICA_NAME
			}			
			hidden_effect = {
				change_country_adjective = VANDALIA_AFRICA_ADJECTIVE
				change_country_tag = VAN
				change_country_color = vandalia_africa_color
				change_country_flag = VANDALIA_AFRICA_FLAG				
			}
			capital_scope = {
				establish_city_effect = yes
				capital_formable_medium_effect = yes
				formable_capital_modifier_normal_effect = yes
			}
			add_3_free_province_investments = yes
		}
		
		ai_will_do = {
			base = 1
		}
	}

}
