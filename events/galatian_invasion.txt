galatian_invasion.1 = { # Barbarian Rumblings
	type = country_event
	title = "galatian_invasion.1.t"
	desc = "galatian_invasion.1.desc"

	left_portrait = scope:pella_attacker.current_ruler
	left_portrait = scope:delphi_attacker.current_ruler
	left_portrait = scope:tylis_attacker.current_ruler
	right_portrait = scope:tavium_attacker.current_ruler
	right_portrait = scope:ankyra_attacker.current_ruler
	right_portrait = scope:pessinous_attacker.current_ruler

	picture = celtic_chiefs

	fire_only_once = yes
	trigger = {
		NOT = { has_global_variable = galation_invasion_game_rule }
		NOT = { has_global_variable = celtic_invasion_occurred }
		OR = {
			# The player decision has already checked its own availability.
			has_variable = player_invasion_eligible
			AND = {
				is_diadochus_trigger = yes
				trigger_if = {
					limit = { c:DND = { exists = yes } }
					c:DND = { is_ai = yes }
				}
				trigger_if = {
					limit = { c:SCD = { exists = yes } }
					c:SCD = { is_ai = yes }
				}
				trigger_if = {
					limit = { c:TEC = { exists = yes } }
					c:TEC = { is_ai = yes }
				}
				trigger_if = {
					limit = { c:TOL = { exists = yes } }
					c:TOL = { is_ai = yes }
				}
				trigger_if = {
					limit = { c:TOI = { exists = yes } }
					c:TOI = { is_ai = yes }
				}
				trigger_if = {
					limit = { c:CEL = { exists = yes } }
					c:CEL = { is_ai = yes }
				}
				current_date > 460.1.1
				OR = {
					p:379 = { NOT = { owner = c:MAC } }
					c:SEL = {
						any_owned_province = {
							is_in_region = galatia_region
							count >= 5
						}
					}
					c:EGY = {
						any_owned_province = {
							is_in_region = galatia_region
							count >= 5
						}
					}
					c:TRE = {
						any_owned_province = {
							is_in_region = galatia_region
							count >= 5
						}
					}
					c:MAC = {
						any_owned_province = {
							is_in_region = galatia_region
							count >= 5
						}
					}
					current_date > 475.1.1
				}
			}
		}
	}

	immediate = {
		set_global_variable = { name = celtic_invasion_occurred }

		if = {
			limit = { c:DND = { exists = yes } }
			c:DND = { save_scope_as = pella_attacker }      # Dindaria
		}
		if = {
			limit = { c:SCD = { exists = yes } }
			c:SCD = { save_scope_as = delphi_attacker }     # Scordiscia
		}
		if = {
			limit = { c:CEL = { exists = yes } }
			c:CEL = { save_scope_as = tylis_attacker }
		}
		if = {
			limit = { c:TOI = { exists = yes } }
			c:TOI = { save_scope_as = tavium_attacker }     # Trocmeria
		}
		if = {
			limit = { c:TOL = { exists = yes } }
			c:TOL = { save_scope_as = pessinous_attacker }  # Tolistobogia
		}
		if = {
			limit = { c:TEC = { exists = yes } }
			c:TEC = { save_scope_as = ankyra_attacker }     # Tectosagia
		}
	}

	option = {
		name = galatian_invasion.1.a
	}

	after = {
		if = {
			limit = { scope:pella_attacker = { exists = yes } }
			random_country = {
				limit = { owns = 379 } # Pella
				save_scope_as = pella_defender
			}
			scope:pella_defender = {
				trigger_event = {
					id = galatian_invasion.10
					days = 30
				}
			}
		}
		if = {
			limit = { scope:delphi_attacker = { exists = yes } }
			random_country = {
				limit = { owns = 424 } # Delphi
				save_scope_as = delphi_defender
			}
			scope:delphi_defender = {
				trigger_event = {
					id = galatian_invasion.20
					days = 60
				}
			}
		}
		if = {
			limit = { scope:tylis_attacker = { exists = yes } }
			random_country = {
				limit = { owns = 480 } # Mesembria
				save_scope_as = tylis_defender
			}
			scope:tylis_defender = {
				trigger_event = {
					id = galatian_invasion.30
					days = 75
				}
			}
		}

		if = {
			limit = {
				NOT = { # verifies that special player events will not occur
					any_country = { has_variable = player_invasion_eligible }
				}
			}
			if = {
				limit = { scope:ankyra_attacker = { exists = yes } }
				scope:ankyra_attacker = {
					trigger_event = {
						id = galatian_invasion.5
						days = 89
					}
				}
			}
			else_if = {
				limit = {
					scope:ankyra_attacker = { exists = no }
					scope:tavium_attacker = { exists = yes }
				}
				scope:tavium_attacker = {
					trigger_event = {
						id = galatian_invasion.5
						days = 89
					}
				}
			}
			else_if = {
				limit = {
					scope:ankyra_attacker = { exists = no }
					scope:tavium_attacker = { exists = no }
					scope:pessinous_attacker = { exists = yes }
				}
				scope:pessinous_attacker = {
					trigger_event = {
						id = galatian_invasion.5
						days = 89
					}
				}
			}
		}
		else = {
			if = {
				limit = { has_variable = player_invasion_eligible }
				trigger_event = {
					id = galatian_invasion.2
					days = 15
				}
			}
		}
	}
}

galatian_invasion.2 = { # Volcae Brethren
	type = country_event
	title = "galatian_invasion.2.t"
	desc = "galatian_invasion.2.desc"

	left_portrait = scope:tavium_attacker.current_ruler
	left_portrait = scope:ankyra_attacker.current_ruler
	right_portrait = scope:pessinous_attacker.current_ruler

	picture = celtic_diplomacy

	goto_location = root

	option = {
		name = galatian_invasion.2.a
		custom_tooltip = galatian_invasion.2.a.tt

		add_political_influence = -40
		add_treasury = -20
		current_ruler = { add_martial = 1 }

		set_variable = {
			name = volcae_brethren
			value = 1
		}

		if = {
			limit = {
				any_current_war = {
					any_war_participant = {
						OR = {
							tag = TEC
							tag = TOL
							tag = TOI
						}
					}
				}
			}
			every_current_war = {
				limit = {
					any_war_participant = {
						OR = {
							tag = TEC
							tag = TOL
							tag = TOI
						}
					}
				}
				force_white_peace = THIS
			}
		}
	}

	option = {
		name = galatian_invasion.2.b
		custom_tooltip = galatian_invasion.2.b.tt

		add_political_influence = -15
		add_treasury = -5
		current_ruler = { add_finesse = 1 }

		# The tribes support the migration, but Tectosagia does not pay to
		# settle their existing conflicts before the expedition begins.
		set_variable = {
			name = volcae_brethren
			value = 1
		}
	}

	option = {
		name = galatian_invasion.2.c
		custom_tooltip = galatian_invasion.2.c.tt

		add_political_influence = -5
		add_aggressive_expansion = 2
		current_ruler = { add_charisma = 1 }

		# A public promise of future land is enough to keep the three tribes
		# coordinated, but it advertises our intentions to the Anatolian states.
		set_variable = {
			name = volcae_brethren
			value = 1
		}
	}

	after = {
		trigger_event = {
			id = galatian_invasion.3
			days = 30
		}
	}
}

galatian_invasion.10 = {
	type = country_event
	title = "galatian_invasion.10.t"
	desc = "galatian_invasion.10.desc"

	left_portrait = scope:pella_attacker.current_ruler
	right_portrait = scope:pella_defender.current_ruler
	picture = city_siege
	goto_location = p:379
	fire_only_once = yes

	trigger = {
		owns = 379
	}

	immediate = { # invade
		random_country = {
			limit = { owns = 379 }
			save_scope_as = pella_defender
		}

		if = {
			limit = { exists = scope:pella_attacker }
			scope:pella_attacker = {
				declare_war_with_wargoal = {
					war_goal = conquer_wargoal
					province = p:379
					target = scope:pella_defender
				}
				galatian_horde_effect = {
					LOC = p:384
					INF = 39
					CHA = 10
					FOOD = 50
				}
			}
		}

		random_current_war = {
			limit = {
				any_war_attacker = { this = scope:pella_attacker }
				any_war_defender = { this = scope:pella_defender }
			}
			save_scope_as = first_galatian_war_scope
		}
		c:TEC = {
			trigger_event = { id = galatian_invasion.13 days = 1 }
		}
	}

	option = {
		name = galatian_invasion.10.a
		add_stability = -5
	}

	after = {
		if = {
			limit = {
				NOT = { c:TRE = { owns = 379 } }
			}
			c:TRE = {
				trigger_event = { id = galatian_invasion.11 days = 1 }
			}
		}

		every_country = {
			limit = { country_culture_group = hellenic }
			trigger_event = { id = galatian_invasion.12 days = 3 }
		}
	}
}

galatian_invasion.20 = { # Second Wave
	type = country_event
	title = "galatian_invasion.20.t"
	desc = "galatian_invasion.20.desc"

	left_portrait = scope:delphi_attacker.current_ruler
	right_portrait = scope:delphi_defender.current_ruler
	picture = city_siege
	goto_location = root

	fire_only_once = yes

	trigger = {
		owns = 424
	}

	immediate = { # invade
		random_country = {
			limit = { any_owned_province = { province_id = 424 } }
			save_scope_as = delphi_defender
		}
		if = {
			limit = {
				scope:delphi_defender = { NOT = { owns = 475 } }
			}
			random_country = {
				limit = { any_owned_province = { province_id = 475 } }
				save_scope_as = passerby_one
			}
		}
		if = {
			limit = {
				NOR = {
					scope:delphi_defender = { owns = 399 }
					trigger_if = {
						limit = { exists = scope:passerby_one }
						scope:passerby_one = { owns = 399 }
					}
				}
			}
			random_country = {
				limit = { any_owned_province = { province_id = 399 } }
				save_scope_as = passerby_two
			}
		}
		if = {
			limit = {
				NOR = {
					scope:delphi_defender = { owns = 469 }
					trigger_if = {
						limit = { exists = scope:passerby_one }
						scope:passerby_one = { owns = 469 }
					}
				}
			}
			random_country = {
				limit = { any_owned_province = { province_id = 469 } }
				save_scope_as = passerby_three
			}
		}

		scope:delphi_attacker = {
			if = {
				limit = { exists = scope:passerby_one }
				add_military_access = scope:passerby_one
			}
			if = {
				limit = { exists = scope:passerby_two }
				add_military_access = scope:passerby_two
			}
			if = {
				limit = { exists = scope:passerby_three }
				add_military_access = scope:passerby_three
			}
			declare_war_with_wargoal = {
				war_goal = conquer_wargoal
				province = p:424
				target = scope:delphi_defender
			}
			galatian_horde_effect = {
				LOC = p:469
				INF = 39
				CHA = 10
				FOOD = 50
			}
		}

		random_current_war = {
			limit = {
				any_war_attacker = { this = scope:delphi_attacker }
				any_war_defender = { this = scope:delphi_defender }
			}
			save_scope_as = second_galatian_war_scope
		}
		c:TEC = {
			trigger_event = { id = galatian_invasion.23 days = 1 }
		}
	}

	option = {
		name = galatian_invasion.20.a
	}

	after = {
		every_country = {
			limit = {
				capital_scope = {
					OR = {
						is_in_area = boeotia_area
						is_in_area = central_greece_area
						is_in_area = aetolia_area
					}
				}
				NOT = { owns = 424 }
			}
			trigger_event = { id = galatian_invasion.21 days = 1 }
		}
		every_country = {
			limit = { country_culture_group = hellenic }
			trigger_event = { id = galatian_invasion.22 days = 3 }
		}
	}
}

galatian_invasion.107 = { # Chariot Race - End
	type = country_event
	title = "galatian_invasion.107.t"
	desc = "galatian_invasion.107.desc"
	picture = "celtic_village"

	right_portrait = c:TEC.current_ruler
	right_portrait = c:TOL.current_ruler
	right_portrait = c:TOI.current_ruler

	immediate = {
		if = {
			limit = {
				any_character = { this = scope:narrow_victorious }
			}
			scope:narrow_victorious = {
				save_scope_as = chariot_victor
				award_tournament_points_effect = { POINTS = 3 }
			}
		}
		else_if = {
			limit = {
				any_character = { this = scope:greatly_victorious }
			}
			scope:greatly_victorious = {
				save_scope_as = chariot_victor
				award_tournament_points_effect = { POINTS = 4 }
			}
		}
		else = {
			random_character = {
				limit = {
					OR = {
						this = scope:tourney_participant_1
						this = scope:tourney_participant_2
						this = scope:tourney_participant_3
					}
					NOT = { this = scope:scared_rider }
					NOT = { this = scope:disaster }
				}
				weight = {
					base = 1
					modifier = {
						factor = 1
						martial > 2
					}
					modifier = {
						factor = 2
						martial > 4
					}
					modifier = {
						factor = 2
						finesse > 4
					}
					modifier = {
						factor = 4
						martial > 8
					}
				}
				save_scope_as = chariot_victor
				award_tournament_points_effect = { POINTS = 3 }
			}
		}
	}

	option = {
		name = galatian_invasion.107.a
		trigger_event = { id = galatian_invasion.110 }
	}
}
