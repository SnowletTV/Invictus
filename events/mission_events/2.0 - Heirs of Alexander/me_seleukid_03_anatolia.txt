        			name = {
        				name = "CAP"
        			}
        			change_country_tag = CAP

        			change_country_color = cappadocian_color

    		}
			scope:ariobarzanes_scope = {
				move_country = scope:cappadocia_country
				remove_character_modifier = foreign_citizen

			}
			scope:cappadocia_country = {
				create_family = scope:ariobarzanes_scope
				set_as_ruler = scope:ariobarzanes_scope

				target = scope:cappadocia_country

    			}
    			custom_tooltip = me_seleukid_03_anatolia.10.a.2.tt
    			hidden_effect = {
    				every_owned_province = {
    					limit = {
    						is_in_region = cappadocia_pontica_region
    					}
    					set_owned_by = scope:cappadocia_country
    				}
    			}
    		}
        }
    }
    option = {
        ai_chance = { factor = 5 }
        name = me_seleukid_03_anatolia.10.b
        add_stability = 3
    }
