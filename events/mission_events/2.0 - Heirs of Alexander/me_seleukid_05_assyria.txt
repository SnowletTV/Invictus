me_seleukid_05_assyria.120 = { # Task 12 - King of Sophene
    type = country_event
    title = me_seleukid_05_assyria.120.t
    desc = me_seleukid_05_assyria.120.desc
    picture = diadochi_greek_royal_wedding
    goto_location = capital_scope

    left_portrait = current_ruler
    right_portrait = scope:seleucid_overlord.current_ruler

    option = { # accept the match
        name = me_seleukid_05_assyria.120.a
        ai_chance = { factor = 65 }
        scope:seleucid_overlord = { set_variable = sophene_yes }
        hidden_effect = {
            if = { # any prior subjection that is not already a satrapy under our new overlord is dissolved
                limit = {
                    is_subject = yes
                    OR = {
                        NOT = { overlord = scope:seleucid_overlord }
                        NOT = { is_subject_type = satrapy }
                    }
                }
                overlord = { release_subject = ROOT }
            }
            current_ruler = { # divorce
                if = {
                    limit = { is_married = yes }
                    divorce_character = THIS.spouse
                }
            }
            create_character = {
                female = yes
                culture = scope:seleucid_overlord.current_ruler.culture
                religion = scope:seleucid_overlord.current_ruler.religion
                family = scope:seleucid_overlord.current_ruler.family
                force_add_trait = seleucids
                age = 20
                save_scope_as = seleucid_princess_scope
            }
        }
        show_as_tooltip = { # sealed when the envoy carries word home
            current_ruler = { marry_character = scope:seleucid_princess_scope }
            add_opinion = { target = scope:seleucid_overlord modifier = sel_loyalty_earned }
            add_stability = 10
            scope:seleucid_overlord = {
                make_subject = {
                    type = satrapy
                    target = root
                }
            }
        }
    }
    option = {
        name = me_seleukid_05_assyria.120.b
        ai_chance = { factor = 35 }
        trigger = { NOT = { is_subject_of = scope:seleucid_overlord } }
    }
    after = {
        scope:seleucid_overlord = { trigger_event = { id = me_seleukid_05_assyria.1201  days = 1 } }
    }
}
