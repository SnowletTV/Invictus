	right_portrait = c:MAC.current_ruler

		c:MAC = {
			random_current_war = {
				limit = { any_war_participant = { tag = PRY } }
				save_scope_as = mac_succession_war
			}

# Hidden watcher - started by achaean_mission_02_task on_completion, re-schedules itself until the
# ruler is personally wealthy while Rapid Urbanization is active, then fires me_achaean_missions.12 once.
me_achaean_missions.13 = {
	type = country_event
	hidden = yes

	trigger = {
		has_country_modifier = rapid_urbanization
	}

	immediate = {
		if = {
			limit = {
				current_ruler = { wealth >= 150 }
				NOT = { has_variable = achaean_ruler_donation_cooldown }
			}
			trigger_event = { id = me_achaean_missions.12 }
		}
		else = {
			trigger_event = {
				id = me_achaean_missions.13
				days = { 90 180 }
			}
		}
	}
}

# Flavor - the ruler may self-fund the League's urbanization from his own personal coffers.

	picture = greek_marketplace
	left_portrait = current_ruler

	trigger = {

		current_ruler = { wealth >= 150 }
	}

	# Donate 75 gold - treasury +75 gold, +20 PI, +10 corruption, ruler gains Benefactor; Oligarchs -20 approval

		}
		add_treasury = 75
		add_political_influence = 20
		add_party_approval = {
			party_type = oligarchic_party
			value = -20
		}
		ai_chance = { factor = 1 }
	}

	# Donate 25 gold - treasury +25 gold
	option = {
		name = me_achaean_missions_13.b
		current_ruler = { add_gold = -25 }
		add_treasury = 25
		ai_chance = { factor = 5 }
	}

	# Donate nothing
	option = {
		name = me_achaean_missions_13.c
		ai_chance = { factor = 10 }
	}

	after = {
		set_variable = {
			name = achaean_ruler_donation_cooldown
			days = 3650
		}
	}
}

# Hidden watcher - started by achaean_mission_01 on_start. Re-schedules itself; when an Achaean athlete
# has won an Olympic victory (some country holds the olympic_victor opinion of us and we have a
# not-yet-celebrated gold medallist), fires me_achaean_missions.15 once for that character.
me_achaean_missions.14 = {
	type = country_event
	hidden = yes

	trigger = { tag = ACL }

	immediate = {
		if = {
			limit = {
				any_country = {
					has_opinion = {
						modifier = olympic_victor
						target = ROOT
					}
				}
				any_character = {
					is_alive = yes
					has_variable = number_of_olympic_wins
					NOT = { has_variable = achaean_curse_broken_personal }
				}
			}
			random_character = {
				limit = {
					is_alive = yes
					has_variable = number_of_olympic_wins
					NOT = { has_variable = achaean_curse_broken_personal }
				}
				save_scope_as = olympic_victor_char
				set_variable = achaean_curse_broken_personal
			}
			trigger_event = { id = me_achaean_missions.15 }
		}
		trigger_event = {
			id = me_achaean_missions.14
			days = { 90 180 }
		}
	}
}

							}
						}
						add_claim = scope:achaean_offended_suzerain
					}
				}
			}
			if = {
				limit = { is_ai = no }
				# a human keeps playing - join as a feudatory instead of being annexed
				c:ACL = { make_subject = { target = ROOT  type = feudatory } }
				c:ACL = { trigger_event = { id = me_achaean_missions.24 } }
			}
			else = {
				c:ACL = { add_aggressive_expansion = root.num_of_provinces }
				every_owned_province = { set_owned_by = c:ACL }
				c:ACL = { trigger_event = { id = me_achaean_missions.21 } }
			}
		}
	}

	option = {		### Reject terms
		name = me_achaean_missions_20.b
		ai_chance = { factor = 100 }

		save_scope_as = achaean_rejecting_country
		hidden_effect = {
			every_owned_province = { add_claim = c:ACL }
			c:ACL = { trigger_event = { id = me_achaean_missions.22 } }
		}
	}

			}

				limit = {
					exists = c:SPA

						any_owned_province = {
							OR = {
								is_in_area = south_peleponnesos_area
								is_in_area = arcadia_area
							}
						}
					}
				}
				every_owned_province = { add_claim = c:SPA }
			}
			c:ACL = { trigger_event = { id = me_achaean_missions.24 } }
		}
	}

	option = {		### Reject terms
		name = me_achaean_missions_23.b
		ai_chance = { factor = 100 }

		save_scope_as = achaean_rejecting_country
		hidden_effect = {
			area:arcadia_area = {
				every_area_province = {
					limit = { owner = scope:achaean_rejecting_country }
					add_claim = c:ACL
				}
			}
			c:ACL = { trigger_event = { id = me_achaean_missions.25 } }
		}
	}
