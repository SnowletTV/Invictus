# Event 6: The Hinterland Revolt (Civil War Spawns)
paris_mission.6 = {
    type = country_event
    title = "paris_mission.6.t" 
    desc = "paris_mission.6.desc"
    picture = looting
    
    option = { 
        name = "paris_mission.6.a" # "To arms!"
        hidden_effect = {
            # 1. Target a rural, rough province in Senonia
            random_owned_province = {
                limit = {
                    is_in_area = senonia_area
                    has_city_status = no
                    OR = { terrain = forest terrain = hills terrain = mountain }
                }
                save_scope_as = rebel_capital
            }
            
            # 2. Spawn the revolt
            scope:rebel_capital = {
                create_country = {
                    name = { name = parisii_revolt_name adjective = parisii_revolt_name_adjective }
                    change_country_color = dark_rust
                    change_government = tribal_kingdom
                    set_primary_culture = sennonian
                    set_country_religion = druidism
                    set_country_heritage = default_heritage
                    save_scope_as = parisii_revolt_scope
                }
                set_owned_by = scope:parisii_revolt_scope
                
                # 3. Pull adjacent wild territories into the revolt
                every_neighbor_province = {
                    limit = {
                        owner = root
                        has_city_status = no
                        is_in_area = senonia_area
                        OR = { terrain = forest terrain = hills terrain = mountain }
                    }
                    set_owned_by = scope:parisii_revolt_scope
                }
            }
            
            # 4. Declare War
            c:PRS = {
                declare_war_with_wargoal = {
                     war_goal = conquer_wargoal 
                     province = scope:rebel_capital 
                     target = scope:parisii_revolt_scope
                }
                # 5. Set the flag so the UI knows the war actually happened
                set_global_variable = { name = old_guard_revolt_fired value = 1 }
            }
        }
    }
}

# Event 10: The Confederation Finale
paris_mission.10 = {
    type = country_event
    title = "paris_mission.10.t"
    desc = "paris_mission.10.desc"
    picture = throneroom
    left_portrait = current_ruler
    
    trigger = { always = yes }
    
    option = {
        name = "paris_mission.10.a"
        
        # Flags the state for your Triggered Modifiers
        set_variable = { name = seine_confederation_formed value = 1 }
        
        add_innovation = 8
        add_political_influence = 80
        add_treasury = 500
        add_centralization = 15
        
        if = {
            limit = { var:seine_forts_built >= 3 }
            add_country_modifier = { name = seine_engineering_tier_3 duration = 3650 }
            add_country_modifier = { name = seine_levies_tier_3 duration = 1825 }
        }
        else_if = {
            limit = { var:seine_forts_built = 2 }
            add_country_modifier = { name = seine_engineering_tier_2 duration = 3650 }
            add_country_modifier = { name = seine_levies_tier_2 duration = 1825 }
        }
        else_if = {
            limit = { var:seine_forts_built = 1 }
            add_country_modifier = { name = seine_engineering_tier_1 duration = 3650 }
            add_country_modifier = { name = seine_levies_tier_1 duration = 1825 }
        }

        p:2475 = { add_province_modifier = { name = parisii_confederation_capital duration = -1 } }
        
        current_ruler = {
            add_trait = founder
            add_nickname = NICKNAME_SEINE_FOUNDER
            add_martial = 1
            add_finesse = 1
            add_charisma = 1
            add_zeal = 1
        }
        change_country_name = SEINE_CONFEDERATION
        
        # Instant Law Rewards (The permanent stuff is handled by the Modifiers)
        if = {
            limit = { 
                OR = { has_law = increased_council_powers is_republic = yes }
            }
            add_stability = 15
        }
        else_if = {
            limit = { 
                OR = { has_law = absolute_authority is_monarchy = yes }
            }
            current_ruler = { add_popularity = 25 }
        }
    }
}

# Event 100: Pulse for Urbanized Hubs
paris_mission.100 = {
    type = country_event
    hidden = yes
    trigger = {
        any_owned_province = {
            has_province_modifier = parisii_protected_trade_hub
            has_city_status = yes
        }
    }
    immediate = {
        every_owned_province = {
            limit = {
                has_province_modifier = parisii_protected_trade_hub
                has_city_status = yes
            }
            remove_province_modifier = parisii_protected_trade_hub
            add_province_modifier = { name = parisii_urban_crossing duration = -1 }
            trigger_event = { id = paris_mission.7 } 
        }
    }
}

paris_mission.101 = {
    type = country_event
    hidden = yes

    trigger = {
        has_variable = seine_confederation_formed
    }

    immediate = {
        # Check and apply/remove Council Harmony - works
        if = {
            limit = {
                OR = {
                    is_republic = yes
                    has_law = increased_council_powers 
                }
            }
            add_country_modifier = {
                name = seine_council_harmony
                duration = -1 # -1 means it lasts until manually removed
            }
        }
        else = {
            remove_country_modifier = seine_council_harmony
        }

        # Check and apply/remove Autocratic Grip
        if = {
            limit = {
                OR = { 
                    is_monarchy = yes 
                    has_law = absolute_authority 
                }
            }
            add_country_modifier = {
                name = seine_autocratic_grip
                duration = -1
            }
        }
        else = {
            remove_country_modifier = seine_autocratic_grip
        }
    }
}
