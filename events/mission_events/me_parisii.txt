# me_parisii.5 - The Sennones Question & Assurances (Parisii Player decides)
me_parisii.5 = {
    type = country_event
    title = me_parisii.5.t
    desc = me_parisii.5.desc
    picture = celtic_diplomacy
    left_portrait = current_ruler
    right_portrait = c:SNN.current_ruler

    trigger = {
        exists = c:SNN
        c:SNN = { is_subject = no }
    }

    # Option A: State Treasury Assurances
    option = {
        name = me_parisii.5.a
        add_treasury = -30
        add_political_influence = -15
        current_ruler = { add_popularity = -10 }
        
        # Discounts for persuasive rulers
        if = {
            limit = {
                current_ruler = {
                    OR = {
                        has_trait = orator
                        has_trait = charming
                        has_trait = silver_tongued
                    }
                }
            }
            add_political_influence = 10
            current_ruler = { add_popularity = 5 }
        }

        set_variable = promised_sennonian_family_assurances
        c:SNN = {
            set_variable = sennonian_assurances_granted
            trigger_event = { id = me_parisii.47 days = { 3 5 } }
        }
    }

    # Option B: Ruler's Personal Wealth (Wealthy Ruler)
    option = {
        name = me_parisii.5.b
        trigger = {
            current_ruler = {
                wealth >= 50
                NOT = { has_trait = selfish has_trait = miserly }
            }
        }
        current_ruler = {
            add_gold = -50
            add_friend = c:SNN.current_ruler
        }
        set_variable = promised_sennonian_family_assurances
        c:SNN = {
            set_variable = sennonian_assurances_granted
            trigger_event = { id = me_parisii.47 days = { 3 5 } }
        }
    }

    # Option C: Refuse Assurances
    option = {
        name = me_parisii.5.c
        set_variable = SNN_bolster_OldGuard
        c:SNN = {
            add_opinion = {
                modifier = sennonian_refused_assurances
                target = root
            }
            set_variable = refused_sennonian_assurances
            trigger_event = { id = me_parisii.47 days = { 3 5 } }
        }
    }
}

# me_parisii.47 - Sennones Fealty Decision (SNN AI receives this)
me_parisii.47 = {
    type = country_event
    title = me_parisii.47.t
    desc = me_parisii.47.desc
    picture = council_country
    left_portrait = current_ruler # SNN Ruler
    right_portrait = global_var:parisii_unification_leader.current_ruler # Parisii Ruler

    option = {
        name = me_parisii.47.a # Submit to Lutetia
        ai_chance = {
            base = 25
            modifier = {
                factor = 2.5

            }
            modifier = {
                factor = 2.0
                exists = global_var:parisii_unification_leader
                global_var:parisii_unification_leader = {
                    any_subject = { tag = CTL }

                factor = 0.5

                factor = 0.0 # Equal or stronger neighbors refuse if Parisii is isolated

                num_of_cities >= global_var:parisii_unification_leader.num_of_cities
                NOT = {
                    global_var:parisii_unification_leader = {
                        any_subject = { count >= 1 }
                    }
                }
            }
            modifier = {
                factor = 2.0
                exists = global_var:parisii_unification_leader
                opinion = { target = global_var:parisii_unification_leader value >= 100 }

            modifier = {
                factor = 1.5
                exists = global_var:parisii_unification_leader
                opinion = { target = global_var:parisii_unification_leader value >= 50 }
            }
            modifier = {
                factor = 0.0 # Proud/brave/ambitious rulers refuse
                current_ruler = {
                    OR = {
                        has_trait = brave
                        has_trait = ambitious
                        has_trait = proud
                    }
                }
            }
        }
                
        global_var:parisii_unification_leader = {
            make_subject = {
                target = ROOT
                type = feudatory
            }
            trigger_event = { id = me_parisii.48 days = 2 }
        }

        # Guaranteed standing for SNN royal dynasty
        current_ruler.family = {
            add_prestige = 30
            every_family_member = {
                add_prominence = 30
            }
        }

        if = {
            limit = { has_variable = sennonian_assurances_granted }
            add_country_modifier = {
                name = sennonian_assurances_modifier
                duration = -1
            }
        }
    }

        name = me_parisii.47.b # Fight!
        ai_chance = {
            factor = 50
        }
        global_var:parisii_unification_leader = {
            trigger_event = { id = me_parisii.49 days = 2 }
        }
    }
}

# me_parisii.9 - The Traditionalist Challenge (Fires on Task 5 start)
me_parisii.9 = {
    type = country_event
    title = me_parisii.9.t
    picture = council_country
    left_portrait = current_ruler
    right_portrait = scope:traditionalist_spokesperson

    desc = {
        first_valid = {
            triggered_desc = {
                trigger = {
                    has_variable = dispossessed_sennonian_noble_var
                    exists = scope:traditionalist_spokesperson
                    scope:traditionalist_spokesperson = var:dispossessed_sennonian_noble_var
                }
                desc = me_parisii.9.desc_sennonian_exile
            }
            triggered_desc = {
                trigger = {
                    scope:traditionalist_spokesperson = {
                        is_clan_chief = yes
                    }
                }
                desc = me_parisii.9.desc_clan_chief
            }
            triggered_desc = {
                trigger = {
                    scope:traditionalist_spokesperson = {
                        has_variable = is_generated_hero
                    }
                }
                desc = me_parisii.9.desc_hometown_hero
            }
            triggered_desc = {
                trigger = {
                    always = yes
                }
                desc = me_parisii.9.desc_minor_noble
            }
        }
    }

    immediate = {
        # Priority 1: Check for dispossessed Sennonian noble if SNN was absorbed without assurances
        if = {
            limit = {
                has_variable = dispossessed_sennonian_noble_var
                var:dispossessed_sennonian_noble_var = { is_alive = yes }
            }
            var:dispossessed_sennonian_noble_var = { save_scope_as = traditionalist_spokesperson }
        }
        # Fallback 1: Look for disloyal traditionalist clan chiefs
        if = {
            limit = { NOT = { exists = scope:traditionalist_spokesperson } }
            random_character = {
                limit = {
                    is_alive = yes
                    is_clan_chief = yes
                    is_ruler = no
                    loyalty < 50
                }
                save_scope_as = traditionalist_spokesperson
            }
        }
        # Fallback 2: Look for ambitious minor nobles with a power base
        if = {
            limit = { NOT = { exists = scope:traditionalist_spokesperson } }
            random_character = {
                limit = {
                    is_alive = yes
                    is_clan_chief = no
                    is_ruler = no
                    OR = {
                        loyalty < 50
                        power_base >= 10
                    }
                }
                save_scope_as = traditionalist_spokesperson
            }
        }
        # Fallback 3: Generate a commoner/hometown hero from the backwater Sennonian core
        if = {
            limit = { NOT = { exists = scope:traditionalist_spokesperson } }
            create_character = {
                culture = sennonian
                religion = druidism
                female = no
                age = 30
                add_trait = brave
                add_trait = ambitious
                add_trait = zealous
                add_trait = stubborn
                save_scope_as = traditionalist_spokesperson
            }
            scope:traditionalist_spokesperson = {
                set_variable = is_generated_hero
                add_loyalty = scorned_claimant_loyalty
            }
        }

        # Save traditionalist_spokesperson as a country variable to persist across event borders safely
        set_variable = { name = traditionalist_spokesperson_var value = scope:traditionalist_spokesperson }
    }    

    option = {
        name = me_parisii.9.a # Buy loyalty via concessions
        add_stability = -5
        current_ruler = {
            add_popularity = -15
        }
        add_political_influence = -25
        add_centralization = -5
        trigger_event = { id = me_parisii.10 days = 14 }
    }

    option = {
        name = me_parisii.9.b # Refuse and prepare for war!
        add_centralization = 5
        trigger_event = { id = me_parisii.11 days = 4 }
    }
}

                limit = { p:3543 = { exists = owner } }
                p:3543 = {
                    save_scope_as = southern_trade_port
                    owner = { save_scope_as = southern_trade_partner }
                }
            }
        }
    }

    # Option A: Accept Mediterranean exchange
    option = {
        name = me_parisii.37.a
        custom_tooltip = me_parisii.37.a.tt
        # Deforest wood to plant the premier vines of the Côte-d'Or
        p:2435 = {
            set_trade_goods = wine
        }
        # Grant Paris capital investment to import the wine
        p:2475 = {
            add_1_free_province_investments = yes
        }
        # Transfer captive slaves to southern trade partner if they exist
        if = {
            limit = {
                exists = scope:southern_trade_partner
            }
            scope:southern_trade_partner = {
                trigger_event = {
                    id = me_parisii.46
                    days = { 7 14 }
                }
            }
        }
        # Transfer captive slaves to Massalia if they exist
        if = {
            limit = {
                any_owned_province = {
                    any_pops_in_province = {
                        pop_type = slaves
                        NOT = { pop_culture_group = gallic }
                    }
                }
            }
            random_owned_province = {
                limit = {
                    any_pops_in_province = {
                        pop_type = slaves
                        NOT = { pop_culture_group = gallic }
                    }
                }
                random_pops_in_province = {
                    limit = {
                        pop_type = slaves
                        NOT = { pop_culture_group = gallic }
                    }
                    kill_pop = yes
                }
                random_pops_in_province = {
                    limit = {
                        pop_type = slaves
                        NOT = { pop_culture_group = gallic }
                    }
                    kill_pop = yes
                }
            }
        }
        else = {
            random_owned_province = {
                limit = {
                    any_pops_in_province = {
                        pop_type = slaves
                    }
                }
                random_pops_in_province = {
                    limit = {
                        pop_type = slaves
                    }
                    kill_pop = yes
                }
            }
        }
        add_treasury = 50
    }

    # Option B: Preserve Traditional Rites
    option = {
        name = me_parisii.37.b
        add_centralization = 5
        current_ruler = { add_popularity = 10 }
    }
