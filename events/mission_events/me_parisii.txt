        if = {

# me_parisii.4 - Fealty Demanded: Peaceful (SNN/CTL AI receives this)
# Dynamic submission evaluation (base + size/opinion/trait multipliers).
me_parisii.4 = {
    type = country_event
    title = me_parisii.4.t
    desc = me_parisii.4.desc
    picture = council_country
    left_portrait = current_ruler # Recipient of Ultimatum
    right_portrait = scope:parisii_player_scope.current_ruler # Parisii

    option = {
        name = me_parisii.4.a # Submit to Lutetia
        ai_chance = {
            base = 35
            modifier = {
                factor = 2.0
                num_of_cities < 5
            }
            modifier = {
                factor = 1.5
                num_of_cities < 10
            }
            modifier = {
                factor = 2.0
                opinion = { target = scope:parisii_player_scope value >= 100 }
            }
            modifier = {
                factor = 1.5
                opinion = { target = scope:parisii_player_scope value >= 50 }
            }
            modifier = {
                factor = 2.5
                has_variable = offered_concessions_var
            }
            modifier = {
                factor = 2.0
                current_ruler = {
                    OR = {
                        has_trait = humble
                        has_trait = submissive
                        has_trait = weak_willed
                    }
                }
            }
            modifier = {
                factor = 0.0 # Proud/brave/ambitious rulers always refuse
                current_ruler = {
                    OR = {
                        has_trait = brave
                        has_trait = ambitious
                        has_trait = proud
                    }
                }
            }
        }
        clear_variable = offered_concessions_var
                
        scope:parisii_player_scope = {
            make_subject = {
                target = ROOT
                type = feudatory
            }
            if = {
                limit = { ROOT = { tag = SNN } }
                trigger_event = { id = me_parisii.6_snn days = 2 }
            }
            if = {
                limit = { ROOT = { tag = CTL } }
                trigger_event = { id = me_parisii.6_ctl days = 2 }
            }
        }
    }

    option = {
        clear_variable = offered_concessions_var
        name = me_parisii.4.b # Fight!
        ai_chance = {
            factor = 55
        }
        scope:parisii_player_scope = {
            if = {
                limit = { ROOT = { tag = SNN } }
                trigger_event = { id = me_parisii.7_snn days = 2 }
            }
            if = {
                limit = { ROOT = { tag = CTL } }
                trigger_event = { id = me_parisii.7_ctl days = 2 }
            }
        }
    }
}

# me_parisii.9 - The Old Guard Ultimatum (Fires on Task 6 start)
me_parisii.9 = {
    type = country_event
    title = me_parisii.9.t
    picture = council_country
    left_portrait = current_ruler
    right_portrait = scope:traditionalist_spokesperson

    desc = {
        first_valid = {
            triggered_desc = {
                trigger = {
                    scope:traditionalist_spokesperson = {
                        is_clan_chief = yes
                    }
                }
                desc = me_parisii.9.desc_clan_chief
            }
            triggered_desc = {
                trigger = {
                    scope:traditionalist_spokesperson = {
                        has_variable = is_generated_hero
                    }
                }
                desc = me_parisii.9.desc_hometown_hero
            }
            triggered_desc = {
                trigger = {
                    always = yes
                }
                desc = me_parisii.9.desc_minor_noble
            }
        }
    }

    immediate = {
        # Fallback 1: Look for disloyal traditionalist clan chiefs
        random_character = {
            limit = {
                is_alive = yes
                is_clan_chief = yes
                is_ruler = no
                loyalty < 50
            }
            save_scope_as = traditionalist_spokesperson
        }
        # Fallback 2: Look for ambitious minor nobles with a power base
        if = {
            limit = { NOT = { exists = scope:traditionalist_spokesperson } }
            random_character = {
                limit = {
                    is_alive = yes
                    is_clan_chief = no
                    is_ruler = no
                    OR = {
                        loyalty < 50
                        power_base >= 10
                    }
                }
                save_scope_as = traditionalist_spokesperson
            }
        }
        # Fallback 3: Generate a commoner/hometown hero from the backwater Sennonian core
        if = {
            limit = { NOT = { exists = scope:traditionalist_spokesperson } }
            create_character = {
                culture = sennonian
                religion = druidism
                female = no
                age = 30
                add_trait = brave
                add_trait = ambitious
                add_trait = zealous
                loyalty = 30
                save_scope_as = traditionalist_spokesperson
            }
            scope:traditionalist_spokesperson = {
                set_variable = is_generated_hero
            }
        }

        # Save traditionalist_spokesperson as a country variable to persist across event borders safely
        set_variable = { name = traditionalist_spokesperson_var value = scope:traditionalist_spokesperson }
    }    

    option = {
        name = me_parisii.9.a # Buy loyalty via concessions
        add_stability = -5
        current_ruler = {
            add_popularity = -15
        }
        add_political_influence = -25
        add_centralization = -5
        trigger_event = { id = me_parisii.10 days = 14 }
    }

    option = {
        name = me_parisii.9.b # Refuse and prepare for war!
        add_centralization = 5
        trigger_event = { id = me_parisii.11 days = 4 }
    }
}

# me_parisii.11 - Secession War (Old Guard Breaks Away)
# Secedes forest/hill/mountains. Dynamically establishes capital at existing fort.
# Instantly builds a fort in the rebel capital if none exists.
me_parisii.11 = {
    type = country_event
    title = me_parisii.11.t
    desc = me_parisii.11.desc
    picture = looting
    left_portrait = current_ruler
    right_portrait = scope:traditionalist_spokesperson

    immediate = {
        p:2448 = { save_scope_as = rebel_capital }
        var:traditionalist_spokesperson_var = { save_scope_as = traditionalist_spokesperson }
    }

    option = {
        name = me_parisii.11.a # To arms!
        add_stability = -10

        # Secession war logic nested inside Option Click
        set_variable = { name = parisii_old_guard_revolt_fired value = yes }

        # INSTANT FORT FAILSAFE: Ensure the capital tile has a fort
        scope:rebel_capital = {
            if = {
                limit = {
                    fort_level < 1
                }
                add_building_level = fortress_building
            }
        }

        scope:rebel_capital = {
            create_country = {
                name = { 
                    name = "parisii_revolt_name"
                    adjective = "parisii_revolt_name_adjective"
                }
                change_country_flag = REB_FLAG
                change_country_color = dark_rust
                change_government = tribal_chiefdom
                set_country_heritage = default_heritage
                save_scope_as = old_guard_revolt_scope
            }
        }

        # Dynamic secession of undeveloped hinterlands (gated by pop and civ)
        every_owned_province = {
            limit = {
                #owner = ROOT
                has_city_status = no
                OR = {
                    is_in_area = senonia_area
                    province_id = 2486
                    province_id = 2490
                    province_id = 2487
                }
                OR = {
                    total_population < 15
                    civilization_value < 35
                }
            }
            set_owned_by = scope:old_guard_revolt_scope
        }

        # Proportional threat army spawning based on spokesperson class
        if = {
            limit = {
                scope:traditionalist_spokesperson = {
                    is_clan_chief = yes
                }
            }
            # Clan Chief revolt: heavy armor, war mounts (Strongest)
            scope:old_guard_revolt_scope = {
                create_unit = {
                location = p:2448        
                name = "The Old Guard"
                hidden_effect = {
                    sub_unit = heavy_infantry
                    sub_unit = heavy_infantry
                    sub_unit = heavy_infantry
                    sub_unit = light_cavalry
                    sub_unit = light_cavalry
                    sub_unit = light_cavalry
                    sub_unit = light_infantry
                    sub_unit = light_infantry
                    sub_unit = archers
                    sub_unit = archers
                }
                
            }
        }
    }    
        else_if = {
            limit = {
                scope:traditionalist_spokesperson = {
                    has_variable = is_generated_hero
                }
            }
            # Generated commoner hero: light spearmen, no mounts, no heavy armor (Weaker)
            scope:old_guard_revolt_scope = {
                create_unit = {     
                    location = p:2448
                    name = "The Old Guard"
                    hidden_effect = {    
                        sub_unit = light_infantry
                        sub_unit = light_infantry
                        sub_unit = light_infantry
                        sub_unit = light_infantry
                        sub_unit = spearmen
                        sub_unit = spearmen
                        sub_unit = archers
                        sub_unit = archers }
                    }
                }
        }    
        else = {
            # Ambitious minor noble: mixed force, few mounts, moderate spearmen (Middling)
            scope:old_guard_revolt_scope = {
                create_unit = {
                    location = p:2448
                    name = "The Old Guard"
                    hidden_effect = {
                        sub_unit = heavy_infantry
                        sub_unit = light_cavalry
                        sub_unit = light_cavalry
                        sub_unit = light_infantry
                        sub_unit = light_infantry
                        sub_unit = spearmen
                        sub_unit = spearmen
                        sub_unit = archers
                        sub_unit = archers
                    }           
                }
            }
        }    

        # Declare Reclamation War
        declare_war_with_wargoal = {
            war_goal = imperial_conquest_wargoal
            target = scope:old_guard_revolt_scope
        }
    }
}

    picture = road_construction
    left_portrait = current_ruler

    option = {
        name = me_parisii.18.a # The river basin is secure!
        
        # 1. Country-Level Effects (Executed cleanly at TOP Country scope)
        change_variable = { name = seine_forts_built add = 1 }

        if = {
            limit = {
                scope:fortified_crossing_scope = { has_building = fortress_building }
            }
            add_treasury = -25
        }
        else = {
            add_treasury = -50
        }

        # 2. Province-Level Effects (Scoped down cleanly exactly once)
        scope:fortified_crossing_scope = {
            # Apply local trade hub modifiers
            add_province_modifier = { name = parisii_protected_trade_hub duration = -1 }
            state = {
                add_state_modifier = {
                    name = state_improvement_military
                    duration = -1
                    mode = add
                }
            }

            # Safe port/building expansion safeguard for settlements
            if = {
                limit = {
                    free_building_slots < 1
                }
                add_province_modifier = { name = parisii_crossing_expansion_effort duration = 7200 }
            }

            # Build the fortress if it's not already constructed
            if = {
                limit = {
                    free_building_slots >= 1
                }
                add_building_level = fortress_building
                add_province_modifier = { name = crossing_construction_effort duration = 720 }
            }
        }
    }
}
