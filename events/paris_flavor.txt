namespace = paris_setup

# === SETUP EVENT ===
# Hooks into on_startup to seamlessly assign the custom Heritage to the Parisii on day one
paris_setup.1 = {
    type = country_event
    hidden = yes
    immediate = {
        every_country = {
            limit = { 
                tag = PRS
                NOT = { heritage = parisii_heritage } 
            }
            set_country_heritage = parisii_heritage
        }
    }
}
